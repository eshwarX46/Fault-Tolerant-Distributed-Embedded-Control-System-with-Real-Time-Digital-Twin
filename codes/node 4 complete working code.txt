/**
 * @file ESP32_Node4_TFT_Working.ino
 * @brief ESP32 CAN Node 4 - Gateway with TFT Display
 * 
 * This code implements ESP32 as CAN Node 4 which acts as a gateway
 * and monitoring station with graphical TFT display.
 * 
 * ## Features
 * - CAN bus communication at 250KBPS
 * - Graphical TFT display (ILI9341 compatible)
 * - Real-time monitoring of Node 1 (sensor data)
 * - Real-time monitoring of Node 2 (actuator status)
 * - Real-time monitoring of Node 3 (heartbeat)
 * - Automatic fault detection with timeout (4 seconds)
 * - Color-coded status indicators
 * - System health overview
 * - Node 4 heartbeat transmission
 * 
 * ## Hardware Requirements
 * - ESP32 Development Board
 * - CAN transceiver (MCP2551 or similar)
 * - TFT Display (ILI9341, 320x240)
 * - SPI connections for TFT
 * 
 * ## CAN Communication
 * - Receive: Node 1 (ID: 0x101), Node 2 (ID: 0x200), Node 3 (ID: 0x303)
 * - Send: Heartbeat (ID: 0x304)
 * 
 * ## Display Layout
 * - Header: System title
 * - Node 1 Section: Temperature, Current, Accelerometer
 * - Node 2 Section: Motor Speed, Direction, Relay States
 * - Footer: System status (Normal/Fault/Waiting)
 * - Status indicators: Green=OK, Red=Fault, Yellow=Warning
 * 
 * @author CAN Node Project
 */

// ============================================
//                          LIBRARY INCLUDES
// ============================================

#include <CAN.h>          /**< CAN bus communication library */
#include <TFT_eSPI.h>     /**< TFT display library for ESP32 */

// ============================================
//                          CAN PIN DEFINITIONS
// ============================================

#define TX_GPIO_NUM   21  /**< CAN TX pin (GPIO 21) */
#define RX_GPIO_NUM   22  /**< CAN RX pin (GPIO 22) */

// ============================================
//                          TFT OBJECT
// ============================================

TFT_eSPI tft = TFT_eSPI();  /**< TFT display object */

// ============================================
//                          NODE STATUS STRUCTURES
// ============================================

/**
 * @brief Node 1 status structure
 * Contains sensor data from Node 1
 */
struct Node1Status {
  bool active = false;                  /**< Node 1 active flag */
  unsigned long lastDataReceived = 0;   /**< Timestamp of last data */
  bool fault = false;                   /**< Fault detection flag */
  float temp1 = 0;                      /**< Temperature #1 in Celsius */
  float temp2 = 0;                      /**< Temperature #2 in Celsius */
  float current = 0;                    /**< Current in Amperes */
  int16_t accelX = 0;                  /**< Accelerometer X-axis */
};

/**
 * @brief Node 2 status structure
 * Contains actuator states from Node 2
 */
struct Node2Status {
  bool active = false;                  /**< Node 2 active flag */
  unsigned long lastDataReceived = 0;   /**< Timestamp of last data */
  bool fault = false;                   /**< Fault detection flag */
  int motorSpeed = 0;                  /**< Motor speed (0-255) */
  bool motorDirection = true;           /**< Motor direction (true=Forward) */
  bool relay1State = false;             /**< Relay 1 state (true=ON) */
  bool relay2State = false;             /**< Relay 2 state (true=ON) */
};

/**
 * @brief Node 3 status structure
 * Contains heartbeat data from Node 3
 */
struct Node3Status {
  bool active = false;                  /**< Node 3 active flag */
  unsigned long lastHeartbeat = 0;      /**< Timestamp of last heartbeat */
  bool fault = false;                   /**< Fault detection flag */
};

// ============================================
//                          GLOBAL VARIABLES
// ============================================

Node1Status node1;          /**< Node 1 status instance */
Node2Status node2;          /**< Node 2 status instance */
Node3Status node3;          /**< Node 3 status instance */

unsigned long lastDisplayUpdate = 0;  /**< Timestamp of last display update */

// ============================================
//                          COLOR DEFINITIONS
// ============================================

#define COLOR_BG     TFT_BLACK   /**< Background color */
#define COLOR_OK     TFT_GREEN   /**< Status OK color */
#define COLOR_FAULT  TFT_RED     /**< Fault status color */
#define COLOR_WARN   TFT_YELLOW  /**< Warning status color */
#define COLOR_TEXT   TFT_WHITE   /**< Default text color */
#define COLOR_HEADER TFT_CYAN    /**< Header background color */

// ============================================
//                          SETUP
// ============================================

/**
 * @brief Arduino setup function - Initializes TFT and CAN
 * 
 * This function:
 * 1. Initializes serial communication
 * 2. Initializes TFT display with splash screen
 * 3. Initializes CAN bus at 250KBPS
 * 4. Displays ready status
 */
void setup() {
  Serial.begin(115200);    /**< Initialize serial at 115200 baud */
  delay(1000);             /**< Wait for serial monitor to open */
  
  Serial.println("=================================");
  Serial.println("ESP32 - NODE 4 (GATEWAY + TFT)");
  Serial.println("=================================");
  
  // ============================================
  // INITIALIZE TFT DISPLAY
  // ============================================
  Serial.print("Initializing TFT... ");
  tft.init();                              /**< Initialize TFT */
  tft.setRotation(1);                      /**< Set rotation to landscape */
  tft.fillScreen(COLOR_BG);                /**< Clear screen with black */
  tft.setTextColor(COLOR_TEXT);            /**< Set text color to white */
  tft.setTextSize(2);                      /**< Set text size */
  tft.drawString("Initializing...", 80, 150);  /**< Show splash message */
  Serial.println("OK ✓");
  delay(1000);                             /**< Show splash for 1 second */
  
  // ============================================
  // INITIALIZE CAN BUS
  // ============================================
  Serial.print("Initializing CAN... ");
  CAN.setPins(RX_GPIO_NUM, TX_GPIO_NUM);   /**< Set custom CAN pins */
  
  if (!CAN.begin(250E3)) {                 /**< Start CAN at 250 kbps */
    Serial.println("FAILED!");
    tft.fillScreen(COLOR_FAULT);           /**< Show error on display */
    tft.drawString("CAN ERROR!", 80, 150);
    while (1);                             /**< Halt on error */
  }
  Serial.println("OK ✓");
  
  // Display ready status
  Serial.println("✅ CAN Ready at 250KBPS!");
  Serial.println("Waiting for Node 1, Node 2 and Node 3 data...");
  
  drawDisplay();  /**< Draw initial display */
}

// ============================================
//                          MAIN LOOP
// ============================================

/**
 * @brief Main program loop - Handles CAN and display updates
 * 
 * This function:
 * 1. Receives and processes CAN messages
 * 2. Checks for node timeouts every second
 * 3. Updates display every 200ms
 * 4. Sends Node 4 heartbeat every 2 seconds
 */
void loop() {
  unsigned long now = millis();
  
  // Receive and process CAN messages
  receiveCAN();
  
  // Check for node timeouts (every second)
  static unsigned long lastCheck = 0;
  if (now - lastCheck > 1000) {
    checkTimeouts();
    lastCheck = now;
  }
  
  // Update display (every 200ms)
  if (now - lastDisplayUpdate > 200) {
    drawDisplay();
    lastDisplayUpdate = now;
  }
  
  // Send Node 4 heartbeat (every 2 seconds)
  static unsigned long lastHeartbeat = 0;
  if (now - lastHeartbeat > 2000) {
    sendHeartbeat();
    lastHeartbeat = now;
  }
  
  delay(10);  /**< Small delay to prevent CPU hogging */
}

// ============================================
//                          CAN FUNCTIONS
// ============================================

/**
 * @brief Receive and process CAN messages
 * 
 * Handles incoming CAN messages from:
 * - Node 1 (ID: 0x101) - Sensor data
 * - Node 2 (ID: 0x200) - Status data
 * - Node 3 (ID: 0x303) - Heartbeat
 * 
 * Updates node status structures and logs data to serial.
 */
void receiveCAN() {
  int packetSize = CAN.parsePacket();  /**< Check for available packet */
  
  if (packetSize) {                    /**< If packet received */
    byte data[8];                      /**< Buffer for received data */
    int i = 0;
    while (CAN.available() && i < 8) {
      data[i++] = CAN.read();          /**< Read data bytes */
    }
    
    // ============================================
    // NODE 1 SENSOR DATA (ID: 0x101)
    // ============================================
    if (CAN.packetId() == 0x101) {
      node1.lastDataReceived = millis();
      node1.active = true;
      
      // Clear fault if previously faulted
      if (node1.fault) {
        node1.fault = false;
        Serial.println("✅ Node 1 RECOVERED!");
      }
      
      // Extract sensor data (minimum 8 bytes expected)
      if (i >= 8) {
        node1.temp1 = data[3];                    /**< Temperature #1 */
        node1.temp2 = data[4];                    /**< Temperature #2 */
        node1.current = data[5] / 100.0;          /**< Current (scaled) */
        node1.accelX = (data[6] << 8) | data[7];  /**< Accelerometer X (16-bit) */
      }
      
      // Log data to serial
      Serial.print("Node1 - Temp1:");
      Serial.print(node1.temp1);
      Serial.print("C, Current:");
      Serial.println(node1.current);
    }
    
    // ============================================
    // NODE 2 STATUS DATA (ID: 0x200)
    // ============================================
    else if (CAN.packetId() == 0x200) {
      node2.lastDataReceived = millis();
      node2.active = true;
      
      // Clear fault if previously faulted
      if (node2.fault) {
        node2.fault = false;
        Serial.println("✅ Node 2 RECOVERED!");
      }
      
      // Extract actuator data (minimum 8 bytes expected)
      if (i >= 8) {
        node2.motorSpeed = data[3];                /**< Motor speed */
        node2.motorDirection = (data[4] == 1);     /**< Direction */
        node2.relay1State = (data[5] == 1);        /**< Relay 1 state */
        node2.relay2State = (data[6] == 1);        /**< Relay 2 state */
      }
      
      // Log data to serial
      Serial.print("Node2 - Motor:");
      Serial.print(node2.motorSpeed);
      Serial.println(" RPM");
    }
    
    // ============================================
    // NODE 3 HEARTBEAT (ID: 0x303)
    // ============================================
    else if (CAN.packetId() == 0x303) {
      node3.lastHeartbeat = millis();
      node3.active = true;
      
      // Clear fault if previously faulted
      if (node3.fault) {
        node3.fault = false;
        Serial.println("✅ Node 3 RECOVERED!");
      }
      
      Serial.println("💓 Node 3 Heartbeat received");
    }
  }
}

/**
 * @brief Send Node 4 heartbeat message
 * 
 * Transmits heartbeat over CAN with ID 0x304
 * to indicate Node 4 is alive and monitoring.
 */
void sendHeartbeat() {
  CAN.beginPacket(0x304);              /**< Node 4 Heartbeat ID */
  CAN.write(4);                        /**< Byte 0: Node ID */
  CAN.write(99);                       /**< Byte 1: Message counter */
  CAN.write(millis() / 1000);          /**< Byte 2: Uptime in seconds */
  CAN.write('O');                      /**< Byte 3: 'O' */
  CAN.write('K');                      /**< Byte 4: 'K' */
  CAN.endPacket();                     /**< Send the packet */
}

// ============================================
//                          MONITORING FUNCTIONS
// ============================================

/**
 * @brief Check timeouts for all nodes
 * 
 * Monitors Node 1, Node 2, and Node 3 health by checking
 * time since last received message or heartbeat.
 * Fault is triggered if no data received for 4 seconds.
 */
void checkTimeouts() {
  unsigned long now = millis();
  
  // Node 1 timeout (4 seconds)
  if (node1.active && (now - node1.lastDataReceived > 4000)) {
    if (!node1.fault) {
      node1.fault = true;
      Serial.println("🚨 FAULT! NODE 1 NOT RESPONDING!");
    }
  }
  
  // Node 2 timeout (4 seconds)
  if (node2.active && (now - node2.lastDataReceived > 4000)) {
    if (!node2.fault) {
      node2.fault = true;
      Serial.println("🚨 FAULT! NODE 2 NOT RESPONDING!");
    }
  }
  
  // Node 3 timeout (4 seconds)
  if (node3.active && (now - node3.lastHeartbeat > 4000)) {
    if (!node3.fault) {
      node3.fault = true;
      Serial.println("🚨 FAULT! NODE 3 NOT RESPONDING!");
    }
  }
}

// ============================================
//                          DISPLAY FUNCTIONS
// ============================================

/**
 * @brief Draw the TFT display
 * 
 * Creates a complete graphical interface showing:
 * - System header
 * - Node 1 sensor data with status indicator
 * - Node 2 actuator data with status indicator  
 * - System status footer
 * 
 * Uses color coding: Green=OK, Red=Fault, Yellow=Warning
 */
void drawDisplay() {
  tft.fillScreen(COLOR_BG);  /**< Clear screen */
  
  // ============================================
  // HEADER
  // ============================================
  tft.fillRect(0, 0, 320, 25, COLOR_HEADER);       /**< Header background */
  tft.setTextColor(TFT_BLACK, COLOR_HEADER);       /**< Black text on cyan */
  tft.setTextSize(1);                              /**< Small text size */
  tft.drawString("FAULT-TOLERANT SYSTEM", 160, 8); /**< Center title */
  
  int yOffset = 35;          /**< Starting Y position */
  int lineHeight = 25;       /**< Line spacing */
  
  // ============================================
  // NODE 1 SECTION - SENSOR HUB
  // ============================================
  tft.setTextColor(COLOR_TEXT, COLOR_BG);  /**< White text on black */
  tft.setTextSize(1);
  tft.drawString("NODE 1 - SENSOR HUB", 10, yOffset);
  
  // Status indicator circle
  uint16_t node1Color = node1.fault ? COLOR_FAULT : (node1.active ? COLOR_OK : COLOR_WARN);
  tft.fillCircle(280, yOffset + 5, 5, node1Color);
  
  yOffset += lineHeight;
  tft.drawString("Temp #1: " + String(node1.temp1, 1) + "C", 10, yOffset);
  tft.drawString("Temp #2: " + String(node1.temp2, 1) + "C", 160, yOffset);
  
  yOffset += lineHeight;
  tft.drawString("Current: " + String(node1.current, 2) + "A", 10, yOffset);
  tft.drawString("Accel X: " + String(node1.accelX), 160, yOffset);
  
  // Display fault message if Node 1 is offline
  if (node1.fault) {
    yOffset += lineHeight;
    tft.setTextColor(TFT_YELLOW, COLOR_BG);
    tft.drawString("!!! NODE 1 OFFLINE !!!", 10, yOffset);
    tft.setTextColor(COLOR_TEXT, COLOR_BG);
  }
  
  // ============================================
  // NODE 2 SECTION - ACTUATOR
  // ============================================
  yOffset += lineHeight + 5;
  tft.drawString("NODE 2 - ACTUATOR", 10, yOffset);
  
  // Status indicator circle
  uint16_t node2Color = node2.fault ? COLOR_FAULT : (node2.active ? COLOR_OK : COLOR_WARN);
  tft.fillCircle(280, yOffset + 5, 5, node2Color);
  
  yOffset += lineHeight;
  tft.drawString("Motor Speed: " + String(node2.motorSpeed) + " RPM", 10, yOffset);
  tft.drawString("Direction: " + String(node2.motorDirection ? "FWD" : "REV"), 160, yOffset);
  
  yOffset += lineHeight;
  tft.drawString("Relay 1: " + String(node2.relay1State ? "ON" : "OFF"), 10, yOffset);
  tft.drawString("Relay 2: " + String(node2.relay2State ? "ON" : "OFF"), 160, yOffset);
  
  // Display fault message if Node 2 is offline
  if (node2.fault) {
    yOffset += lineHeight;
    tft.setTextColor(TFT_YELLOW, COLOR_BG);
    tft.drawString("!!! NODE 2 OFFLINE !!!", 10, yOffset);
    tft.setTextColor(COLOR_TEXT, COLOR_BG);
  }
  
  // ============================================
  // FOOTER - SYSTEM STATUS
  // ============================================
  yOffset += lineHeight + 7;
  
  // System Fault triggers if ANY node (1, 2, or 3) has fault
  if (node1.fault || node2.fault || node3.fault) {
    // RED - System fault
    tft.fillRect(5, yOffset, 310, 20, COLOR_FAULT);
    tft.setTextColor(TFT_YELLOW, COLOR_FAULT);
    tft.drawString("SYSTEM FAULT!", 160, yOffset + 7);
  } 
  else if (node1.active && node2.active && node3.active) {
    // GREEN - All nodes active
    tft.fillRect(5, yOffset, 310, 20, COLOR_OK);
    tft.setTextColor(TFT_BLACK, COLOR_OK);
    tft.drawString("SYSTEM NORMAL", 160, yOffset + 7);
  } 
  else {
    // YELLOW - Waiting for nodes
    tft.fillRect(5, yOffset, 310, 20, COLOR_WARN);
    tft.setTextColor(TFT_BLACK, COLOR_WARN);
    tft.drawString("WAITING FOR NODES...", 160, yOffset + 7);
  }
}
