/**
 * @file ESP32_Node3_Monitor_And_Control.ino
 * @brief ESP32 CAN Node 3 - Monitor and Control System
 * 
 * This code implements ESP32 as CAN Node 3 which monitors both Node 1 (sensors)
 * and Node 2 (actuators) while sending control commands to Node 2.
 * 
 * ## Features
 * - CAN bus communication at 250KBPS
 * - Real-time monitoring of Node 1 (sensor data)
 * - Real-time monitoring of Node 2 (actuator status)
 * - Automatic fault detection with timeout (4 seconds)
 * - Visual indicators: Green (Node1), White (Node2), Blue (Fault)
 * - Audible alerts with buzzer
 * - Hardware reset button for clearing faults
 * - Automatic control commands to Node 2 every 10 seconds
 * - Comprehensive serial debugging output
 * 
 * ## Hardware Requirements
 * - ESP32 Development Board
 * - CAN transceiver (MCP2551 or similar)
 * - 3x LEDs (Green, White, Blue) with 220Ω resistors
 * - 1x Active Buzzer
 * - 1x Push Button (Reset)
 * 
 * ## CAN Communication
 * - Receive: Node 1 (ID: 0x101), Node 2 (ID: 0x200)
 * - Send: Heartbeat (ID: 0x303), Commands (ID: 0x401)
 * 
 * ## Command Sequence (Cyclic)
 * 1. Set Motor Speed = 150
 * 2. Set Motor Direction = FORWARD
 * 3. Relay 1 = ON
 * 4. Relay 2 = ON
 * 5. Relay 1 = OFF
 * 6. Relay 2 = OFF
 * 7. Set Motor Speed = 100
 * 8. Set Motor Direction = REVERSE
 * 9. Set Motor Speed = 0 (STOP)
 * 
 * @author CAN Node Project
 */

// ============================================
//                          LIBRARY INCLUDES
// ============================================

#include <CAN.h>          /**< CAN bus communication library */

// ============================================
//                          CAN PIN DEFINITIONS
// ============================================

#define TX_GPIO_NUM   21  /**< CAN TX pin (GPIO 21) */
#define RX_GPIO_NUM   22  /**< CAN RX pin (GPIO 22) */

// ============================================
//                          LED & BUZZER & BUTTON PIN DEFINITIONS
// ============================================

#define GREEN_LED   2     /**< Green LED - Node 1 status indicator */
#define WHITE_LED   4     /**< White LED - Node 2 status indicator */
#define BLUE_LED    15    /**< Blue LED - Fault indicator (blinking) */
#define BUZZER      16    /**< Active buzzer for audible alerts */
#define RESET_BTN   17    /**< Push button for resetting faults */

// ============================================
//                          NODE 1 STATUS STRUCTURE
// ============================================

/**
 * @brief Node 1 status structure
 * Contains all sensor data from Node 1 and health monitoring
 */
struct Node1Status {
  bool active;                      /**< Node 1 active flag */
  unsigned long lastDataReceived;   /**< Timestamp of last data from Node 1 */
  int missedCount;                  /**< Number of missed heartbeats */
  bool fault;                       /**< Fault detection flag */
  float temp1;                      /**< Temperature #1 in Celsius */
  float temp2;                      /**< Temperature #2 in Celsius */
  float current;                    /**< Current in Amperes */
  int16_t accelX;                   /**< Accelerometer X-axis reading */
};

Node1Status node1;                  /**< Node 1 status instance */

// ============================================
//                          NODE 2 STATUS STRUCTURE
// ============================================

/**
 * @brief Node 2 status structure
 * Contains all actuator states from Node 2 and health monitoring
 */
struct Node2Status {
  bool active;                      /**< Node 2 active flag */
  unsigned long lastDataReceived;   /**< Timestamp of last data from Node 2 */
  int missedCount;                  /**< Number of missed heartbeats */
  bool fault;                       /**< Fault detection flag */
  int motorSpeed;                   /**< Motor speed (0-255) */
  bool motorDirection;              /**< Motor direction (true=Forward) */
  bool relay1State;                 /**< Relay 1 state (true=ON) */
  bool relay2State;                 /**< Relay 2 state (true=ON) */
  bool emergencyStop;               /**< Emergency stop flag */
};

Node2Status node2;                  /**< Node 2 status instance */

// ============================================
//                          CONTROL VARIABLES
// ============================================

int sendCount = 0;                  /**< Heartbeat message counter */
int commandIndex = 0;               /**< Current command index in sequence */
unsigned long lastSendTime = 0;     /**< Timestamp of last heartbeat */
unsigned long lastHeartbeatCheck = 0; /**< Timestamp of last timeout check */
unsigned long lastCommandTime = 0;  /**< Timestamp of last command sent */
int buttonPressCount = 0;           /**< Counter for reset button presses */
int sensorDataCount = 0;            /**< Counter for Node 1 data received */
int statusDataCount = 0;            /**< Counter for Node 2 data received */
bool emergencyStopActive = false;   /**< Emergency stop flag for Node 3 */

// ============================================
//                          LED BLINKING VARIABLES
// ============================================

unsigned long lastFaultBlink = 0;   /**< Timestamp of last fault blink */
bool faultBlinkState = false;       /**< Current state of fault LED blink */

// ============================================
//                          SETUP
// ============================================

/**
 * @brief Arduino setup function - Initializes hardware and CAN
 * 
 * This function:
 * 1. Initializes serial communication
 * 2. Initializes LEDs, buzzer, and button
 * 3. Runs startup test sequence
 * 4. Initializes Node 1 and Node 2 status structures
 * 5. Initializes CAN bus at 250KBPS
 * 6. Displays ready status
 */
void setup() {
  Serial.begin(115200);    /**< Initialize serial at 115200 baud */
  delay(1000);             /**< Wait for serial monitor to open */
  
  Serial.println("=================================");
  Serial.println("ESP32 - NODE 3 (MONITOR + CONTROL)");
  Serial.println("Monitoring Node 1 AND Node 2 + Controlling Node 2");
  Serial.println("=================================");
  
  // ============================================
  // INITIALIZE PERIPHERALS
  // ============================================
  pinMode(GREEN_LED, OUTPUT);
  pinMode(WHITE_LED, OUTPUT);
  pinMode(BLUE_LED, OUTPUT);
  pinMode(BUZZER, OUTPUT);
  pinMode(RESET_BTN, INPUT_PULLUP);  /**< Button with internal pull-up */
  
  // Turn off all indicators initially
  digitalWrite(GREEN_LED, LOW);
  digitalWrite(WHITE_LED, LOW);
  digitalWrite(BLUE_LED, LOW);
  digitalWrite(BUZZER, LOW);
  
  // Perform startup test sequence
  startupTest();
  
  // ============================================
  // INITIALIZE NODE 1 STATUS
  // ============================================
  node1.active = false;
  node1.lastDataReceived = 0;
  node1.missedCount = 0;
  node1.fault = false;
  
  // ============================================
  // INITIALIZE NODE 2 STATUS
  // ============================================
  node2.active = false;
  node2.lastDataReceived = 0;
  node2.missedCount = 0;
  node2.fault = false;
  node2.motorSpeed = 0;
  node2.motorDirection = true;
  node2.relay1State = false;
  node2.relay2State = false;
  node2.emergencyStop = false;
  
  // ============================================
  // INITIALIZE CAN BUS
  // ============================================
  CAN.setPins(RX_GPIO_NUM, TX_GPIO_NUM);  /**< Set custom CAN pins */
  
  if (!CAN.begin(250E3)) {                /**< Start CAN at 250 kbps */
    Serial.println("❌ Starting CAN failed!");
    while (1);                            /**< Halt on error */
  }
  
  // Display system status
  Serial.println("✅ CAN Ready at 250KBPS!");
  Serial.println("");
  Serial.println("Monitoring Node 1 (CAN ID: 0x101) and Node 2 (CAN ID: 0x200)...");
  Serial.println("Sending control commands to Node 2 every 10 seconds...");
  Serial.println("=================================\n");
}

// ============================================
//                          MAIN LOOP
// ============================================

/**
 * @brief Main program loop - Handles monitoring and control
 * 
 * This function:
 * 1. Checks reset button
 * 2. Sends heartbeat every 2 seconds
 * 3. Sends control command sequence every 10 seconds
 * 4. Checks for node timeouts every second
 * 5. Updates LED indicators
 * 6. Processes incoming CAN messages
 */
void loop() {
  unsigned long now = millis();
  
  // ============================================
  // CHECK RESET BUTTON
  // ============================================
  if (digitalRead(RESET_BTN) == LOW) {  /**< Button pressed (LOW due to pull-up) */
    buttonPressCount++;
    Serial.print("🔘 RESET BUTTON PRESSED! (");
    Serial.print(buttonPressCount);
    Serial.println(" times)");
    
    // Provide audible feedback
    digitalWrite(BUZZER, HIGH);
    delay(200);
    digitalWrite(BUZZER, LOW);
    
    // Clear all faults
    clearAllFaults();
    delay(500);  /**< Debounce delay */
  }
  
  // ============================================
  // SEND NODE 3 HEARTBEAT
  // ============================================
  if (now - lastSendTime > 2000) {  /**< Every 2 seconds */
    sendMyHeartbeat();
    lastSendTime = now;
  }
  
  // ============================================
  // SEND CONTROL COMMANDS TO NODE 2
  // ============================================
  if (!emergencyStopActive && (now - lastCommandTime > 10000)) {  /**< Every 10 seconds */
    sendControlCommand();
    lastCommandTime = now;
  }
  
  // ============================================
  // CHECK NODE TIMEOUTS
  // ============================================
  if (now - lastHeartbeatCheck > 1000) {  /**< Every second */
    checkNode1Timeout();
    checkNode2Timeout();
    lastHeartbeatCheck = now;
  }
  
  // ============================================
  // UPDATE LEDS
  // ============================================
  updateLEDs();
  
  // ============================================
  // RECEIVE CAN MESSAGES
  // ============================================
  canReceiver();
  
  delay(10);  /**< Small delay to prevent CPU hogging */
}

// ============================================
//                          STARTUP FUNCTIONS
// ============================================

/**
 * @brief Perform startup test sequence
 * 
 * Tests all output devices (LEDs and buzzer) to verify
 * hardware connectivity and functionality.
 */
void startupTest() {
  Serial.println("Running startup test...");
  
  // Test Green LED
  digitalWrite(GREEN_LED, HIGH);
  Serial.println("🟢 Green LED ON");
  delay(500);
  digitalWrite(GREEN_LED, LOW);
  
  // Test White LED
  digitalWrite(WHITE_LED, HIGH);
  Serial.println("⚪ White LED ON");
  delay(500);
  digitalWrite(WHITE_LED, LOW);
  
  // Test Blue LED
  digitalWrite(BLUE_LED, HIGH);
  Serial.println("🔵 Blue LED ON");
  delay(500);
  digitalWrite(BLUE_LED, LOW);
  
  // Test Buzzer
  digitalWrite(BUZZER, HIGH);
  Serial.println("🔊 Buzzer ON");
  delay(300);
  digitalWrite(BUZZER, LOW);
  Serial.println("🔇 Buzzer OFF");
  
  // Test all LEDs together
  digitalWrite(GREEN_LED, HIGH);
  digitalWrite(WHITE_LED, HIGH);
  digitalWrite(BLUE_LED, HIGH);
  Serial.println("🟢⚪🔵 All LEDs ON");
  delay(500);
  digitalWrite(GREEN_LED, LOW);
  digitalWrite(WHITE_LED, LOW);
  digitalWrite(BLUE_LED, LOW);
  
  Serial.println("✅ Startup test complete!\n");
}

// ============================================
//                          CAN FUNCTIONS
// ============================================

/**
 * @brief Send Node 3 heartbeat message
 * 
 * Transmits heartbeat over CAN with ID 0x303
 * to indicate Node 3 is alive and monitoring.
 */
void sendMyHeartbeat() {
  CAN.beginPacket(0x303);              /**< Node 3 Heartbeat ID */
  CAN.write(3);                        /**< Byte 0: Node ID */
  CAN.write(sendCount);                /**< Byte 1: Message counter */
  CAN.write(millis() / 1000);          /**< Byte 2: Uptime in seconds */
  CAN.endPacket();                     /**< Send the packet */
  sendCount++;                         /**< Increment message counter */
}

/**
 * @brief Send control command sequence to Node 2
 * 
 * Cycles through a predefined sequence of commands to demonstrate
 * full control of Node 2's actuators. Sequence repeats every 9 commands.
 */
void sendControlCommand() {
  Serial.println("");
  Serial.println("=========================================");
  Serial.print(">>> Sending Control Command #");
  Serial.print(commandIndex + 1);
  Serial.println(" to NODE 2 <<<");
  
  // Execute command based on current index
  switch(commandIndex) {
    case 0:
      Serial.println("Command: Set Motor Speed = 150");
      sendCommandToNode2(10, 150);  /**< Command 10: Set speed to 150 */
      break;
      
    case 1:
      Serial.println("Command: Set Motor Direction = FORWARD");
      sendCommandToNode2(12, 1);    /**< Command 12: Direction FORWARD */
      break;
      
    case 2:
      Serial.println("Command: Relay 1 = ON");
      sendCommandToNode2(13, 1);    /**< Command 13: Relay 1 ON */
      break;
      
    case 3:
      Serial.println("Command: Relay 2 = ON");
      sendCommandToNode2(14, 1);    /**< Command 14: Relay 2 ON */
      break;
      
    case 4:
      Serial.println("Command: Relay 1 = OFF");
      sendCommandToNode2(13, 0);    /**< Command 13: Relay 1 OFF */
      break;
      
    case 5:
      Serial.println("Command: Relay 2 = OFF");
      sendCommandToNode2(14, 0);    /**< Command 14: Relay 2 OFF */
      break;
      
    case 6:
      Serial.println("Command: Set Motor Speed = 100");
      sendCommandToNode2(10, 100);  /**< Command 10: Set speed to 100 */
      break;
      
    case 7:
      Serial.println("Command: Set Motor Direction = REVERSE");
      sendCommandToNode2(12, 0);    /**< Command 12: Direction REVERSE */
      break;
      
    case 8:
      Serial.println("Command: Set Motor Speed = 0 (STOP)");
      sendCommandToNode2(10, 0);    /**< Command 10: Stop motor */
      break;
  }
  
  // Advance command index, loop back to 0 after last command
  commandIndex++;
  if (commandIndex > 8) {
    commandIndex = 0;
  }
  
  Serial.println(">>> COMMAND SENT <<<");
  Serial.println("=========================================\n");
}

/**
 * @brief Send a command to Node 2
 * @param command Command code to send
 * @param value Value parameter for the command
 * 
 * Formats and transmits a CAN message with ID 0x401
 * containing target node, command, and value.
 */
void sendCommandToNode2(int command, int value) {
  CAN.beginPacket(0x401);          /**< Command ID */
  CAN.write(2);                    /**< Byte 0: Target Node 2 */
  CAN.write(command);              /**< Byte 1: Command code */
  CAN.write(value);                /**< Byte 2: Command value */
  CAN.endPacket();                 /**< Send the packet */
}

// ============================================
//                          CAN RECEIVER
// ============================================

/**
 * @brief Receive and process CAN messages
 * 
 * Handles incoming CAN messages from:
 * - Node 1 (ID: 0x101) - Sensor data
 * - Node 2 (ID: 0x200) - Status data
 * 
 * Updates node status structures and displays data on serial monitor.
 */
void canReceiver() {
  int packetSize = CAN.parsePacket();  /**< Check for available packet */
  
  if (packetSize) {                    /**< If packet received */
    byte data[8];                      /**< Buffer for received data */
    int i = 0;
    while (CAN.available() && i < 8) {
      data[i++] = CAN.read();          /**< Read data bytes */
    }
    
    // ============================================
    // NODE 1 SENSOR DATA (ID: 0x101)
    // ============================================
    if (CAN.packetId() == 0x101) {
      sensorDataCount++;
      node1.lastDataReceived = millis();
      node1.active = true;
      node1.missedCount = 0;  /**< Reset missed count on successful receive */
      
      // Clear fault if previously faulted
      if (node1.fault) {
        node1.fault = false;
        Serial.println("✅ Node 1 has RECOVERED!");
        checkAndClearBuzzer();
      }
      
      // Extract sensor data (minimum 8 bytes expected)
      if (i >= 8) {
        node1.temp1 = data[3];                    /**< Temperature #1 */
        node1.temp2 = data[4];                    /**< Temperature #2 */
        node1.current = data[5] / 100.0;          /**< Current (scaled) */
        node1.accelX = (data[6] << 8) | data[7];  /**< Accelerometer X (16-bit) */
        
        // Display Node 1 sensor data
        Serial.println("");
        Serial.println("=========================================");
        Serial.print("📥 NODE 1 DATA #");
        Serial.println(sensorDataCount);
        Serial.println("=========================================");
        Serial.print("🌡️ Temperature #1: ");
        Serial.print(node1.temp1);
        Serial.println(" °C");
        Serial.print("🌡️ Temperature #2: ");
        Serial.print(node1.temp2);
        Serial.println(" °C");
        Serial.print("⚡ Current: ");
        Serial.print(node1.current);
        Serial.println(" A");
        Serial.print("📳 Accel X: ");
        Serial.println(node1.accelX);
        Serial.println("=========================================");
        Serial.println("");
      }
    }
    
    // ============================================
    // NODE 2 STATUS DATA (ID: 0x200)
    // ============================================
    else if (CAN.packetId() == 0x200) {
      statusDataCount++;
      node2.lastDataReceived = millis();
      node2.active = true;
      node2.missedCount = 0;  /**< Reset missed count on successful receive */
      
      // Clear fault if previously faulted
      if (node2.fault) {
        node2.fault = false;
        Serial.println("✅ Node 2 has RECOVERED!");
        checkAndClearBuzzer();
      }
      
      // Extract actuator data (minimum 8 bytes expected)
      if (i >= 8) {
        node2.motorSpeed = data[3];                 /**< Motor speed */
        node2.motorDirection = (data[4] == 1);      /**< Direction */
        node2.relay1State = (data[5] == 1);         /**< Relay 1 state */
        node2.relay2State = (data[6] == 1);         /**< Relay 2 state */
        node2.emergencyStop = (data[7] == 1);       /**< Emergency stop */
        
        // Display Node 2 status data
        Serial.println("");
        Serial.println("=========================================");
        Serial.print("📥 NODE 2 STATUS #");
        Serial.println(statusDataCount);
        Serial.println("=========================================");
        Serial.print("🎯 Motor Speed: ");
        Serial.println(node2.motorSpeed);
        Serial.print("🔄 Motor Direction: ");
        Serial.println(node2.motorDirection ? "FORWARD" : "REVERSE");
        Serial.print("🔴 Relay 1: ");
        Serial.println(node2.relay1State ? "ON" : "OFF");
        Serial.print("🔴 Relay 2: ");
        Serial.println(node2.relay2State ? "ON" : "OFF");
        Serial.print("🚨 Emergency Stop: ");
        Serial.println(node2.emergencyStop ? "ACTIVE" : "INACTIVE");
        Serial.println("=========================================");
        Serial.println("");
      }
    }
  }
}

// ============================================
//                          MONITORING FUNCTIONS
// ============================================

/**
 * @brief Check Node 1 timeout
 * 
 * Monitors Node 1 health by checking time since last received message.
 * If no data received for 4 seconds, triggers fault condition.
 */
void checkNode1Timeout() {
  unsigned long now = millis();
  
  if (node1.active && (now - node1.lastDataReceived > 4000)) {  /**< 4 second timeout */
    node1.missedCount++;
    
    if (node1.missedCount >= 1 && !node1.fault) {
      node1.fault = true;
      Serial.println("");
      Serial.println("=========================================");
      Serial.println("🚨 FAULT DETECTED! NODE 1 NOT RESPONDING!");
      Serial.println("=========================================");
      Serial.println("");
      
      digitalWrite(BUZZER, HIGH);  /**< Activate buzzer */
    }
  }
}

/**
 * @brief Check Node 2 timeout
 * 
 * Monitors Node 2 health by checking time since last received message.
 * If no data received for 4 seconds, triggers fault condition.
 */
void checkNode2Timeout() {
  unsigned long now = millis();
  
  if (node2.active && (now - node2.lastDataReceived > 4000)) {  /**< 4 second timeout */
    node2.missedCount++;
    
    if (node2.missedCount >= 1 && !node2.fault) {
      node2.fault = true;
      Serial.println("");
      Serial.println("=========================================");
      Serial.println("🚨 FAULT DETECTED! NODE 2 NOT RESPONDING!");
      Serial.println("=========================================");
      Serial.println("");
      
      digitalWrite(BUZZER, HIGH);  /**< Activate buzzer */
    }
  }
}

//                          INDICATOR FUNCTIONS
// ============================================

/**
 * @brief Update LED indicators
 * 
 * - Green LED: Node 1 active and no fault
 * - White LED: Node 2 active and no fault
 * - Blue LED: Blinking when any fault detected
 */
void updateLEDs() {
  // Green LED - Node 1 status
  digitalWrite(GREEN_LED, node1.active && !node1.fault);
  
  // White LED - Node 2 status
  digitalWrite(WHITE_LED, node2.active && !node2.fault);
  
  // Blue LED - Blinking when ANY fault detected
  if (node1.fault || node2.fault) {
    if (millis() - lastFaultBlink > 500) {  /**< Blink every 500ms */
      faultBlinkState = !faultBlinkState;
      digitalWrite(BLUE_LED, faultBlinkState);
      lastFaultBlink = millis();
    }
  } else {
    digitalWrite(BLUE_LED, LOW);
    faultBlinkState = false;
  }
}

/**
 * @brief Check and clear buzzer if no faults
 * 
 * Automatically turns off buzzer when all faults are cleared.
 */
void checkAndClearBuzzer() {
  if (!node1.fault && !node2.fault) {
    digitalWrite(BUZZER, LOW);
  }
}

// ============================================
//                          FAULT HANDLING
// ============================================

/**
 * @brief Clear all faults and reset monitoring
 * 
 * Resets fault flags, missed counts, and active states.
 * Turns off indicators and prepares for fresh monitoring.
 */
void clearAllFaults() {
  node1.fault = false;
  node1.missedCount = 0;
  node1.active = false;
  
  node2.fault = false;
  node2.missedCount = 0;
  node2.active = false;
  
  // Turn off indicators
  digitalWrite(BLUE_LED, LOW);
  digitalWrite(BUZZER, LOW);
  faultBlinkState = false;
  emergencyStopActive = false;
  
  Serial.println("✅ All faults cleared! Waiting for Node 1 and Node 2 data...");
}
